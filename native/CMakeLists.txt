cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(glovego_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
add_library(glovego_vision SHARED src/depth.cpp)
if(MSVC)
  set_property(TARGET glovego_vision PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(ANDROID)
  target_link_options(glovego_vision PRIVATE "LINKER:-z,max-page-size=16384")
endif()
target_link_libraries(glovego_vision PRIVATE ${OpenCV_LIBS})
target_include_directories(glovego_vision PRIVATE ${OpenCV_INCLUDE_DIRS})
